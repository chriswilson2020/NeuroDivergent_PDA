cmake_minimum_required(VERSION 3.16)
project(PocketPDAInputTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_CONF_SKIP ON CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_subdirectory("${ROOT}/.pio/libdeps/tlora_pager/lvgl" lvgl)
target_compile_definitions(lvgl PUBLIC LV_FONT_MONTSERRAT_20=1)
target_compile_definitions(lvgl PUBLIC LV_FONT_MONTSERRAT_12=1)
add_executable(test_input_navigation "${ROOT}/tests/test_input_navigation.cpp" "${ROOT}/src/hardware/InputManager.cpp")
target_include_directories(test_input_navigation PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${ROOT}/src")
target_link_libraries(test_input_navigation PRIVATE lvgl)
add_executable(test_focus_recovery "${ROOT}/tests/test_focus_recovery.cpp" "${ROOT}/src/core/AppManager.cpp"
    "${ROOT}/src/core/NotificationManager.cpp" "${ROOT}/src/core/EventBus.cpp" "${ROOT}/src/ui/Theme.cpp")
target_include_directories(test_focus_recovery PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${ROOT}/src")
target_link_libraries(test_focus_recovery PRIVATE lvgl)
enable_testing()
add_test(NAME input_navigation COMMAND test_input_navigation)
add_test(NAME focus_recovery COMMAND test_focus_recovery)
add_executable(test_locked_alert_services "${ROOT}/tests/test_locked_alert_services.cpp"
    "${ROOT}/src/core/TimerService.cpp" "${ROOT}/src/core/ReminderService.cpp" "${ROOT}/src/core/OccurrenceLedger.cpp"
    "${ROOT}/src/core/NotificationManager.cpp" "${ROOT}/src/core/EventBus.cpp" "${ROOT}/src/ui/Theme.cpp")
target_include_directories(test_locked_alert_services PRIVATE "${ROOT}/tests/locked_service_stubs" "${ROOT}/tests/power_stubs" "${ROOT}/src")
target_compile_definitions(test_locked_alert_services PRIVATE POCKETPDA_POWER_DIAGNOSTICS=0)
target_link_libraries(test_locked_alert_services PRIVATE lvgl)
add_test(NAME locked_alert_services COMMAND test_locked_alert_services)
if(APPLE)
    add_executable(test_screen_lock "${ROOT}/tests/test_screen_lock.cpp" "${ROOT}/src/data/PasscodeStore.cpp"
        "${ROOT}/src/ui/LockScreen.cpp" "${ROOT}/src/core/NotificationManager.cpp"
        "${ROOT}/src/core/EventBus.cpp" "${ROOT}/src/ui/Theme.cpp")
    target_include_directories(test_screen_lock PRIVATE "${ROOT}/tests/lock_stubs" "${CMAKE_CURRENT_LIST_DIR}" "${ROOT}/src")
    target_link_libraries(test_screen_lock PRIVATE lvgl)
    add_test(NAME screen_lock COMMAND test_screen_lock)
endif()
